# Service Patterns in Axelor

## Overview

Services are the heart of business logic in Axelor applications. They encapsulate domain logic, orchestrate operations across multiple repositories, manage transactions, and handle complex business rules.

## Important: Sequence Management is Automatic

> **CRITICAL: DO NOT write Java code for sequence generation**
>
> Sequences are **automatically handled by Axelor** when you define `sequence="..."` attribute on a string field in the domain XML. You do **NOT** need to write service methods to generate sequence numbers.
>
> **WRONG approach:**
> ```java
> // WRONG - Don't do this!
> public SaleOrder confirm(SaleOrder order) {
>     String orderNumber = Beans.get(SequenceService.class)
>         .getSequenceNumber("sale.order.seq");
>     order.setOrderNumber(orderNumber);
>     return repository.save(order);
> }
> ```
>
> **CORRECT approach:**
> ```xml
> <!-- In domain XML - Axelor handles it automatically -->
> <entity name="SaleOrder">
>   <string name="orderNumber" sequence="sale.order.seq"/>
> </entity>
> ```
> ```java
> // In service - No sequence code needed!
> public SaleOrder confirm(SaleOrder order) {
>     order.setStatusSelect(STATUS_CONFIRMED);
>     return repository.save(order);
>     // orderNumber is automatically generated by Axelor
> }
> ```
>
> **Rare exception:** Only use `JpaSequence.nextValue("seq.name")` if you need the sequence value BEFORE saving the entity. This is extremely rare (< 1% of cases).
>
> **Reference:** See @docs/domains/domain-patterns.md (Sequences section) for complete documentation.

## Service Interface Patterns

### Basic Service Interface

```java
package com.axelor.apps.sale.service;

import com.axelor.apps.sale.db.SaleOrder;
import com.axelor.exception.AxelorException;
import java.math.BigDecimal;
import java.time.LocalDate;

public interface SaleOrderService {

    /**
     * Create a new sale order with default values
     * @param customer The customer for the order
     * @return The created sale order
     * @throws AxelorException if validation fails
     */
    SaleOrder createOrder(Customer customer) throws AxelorException;

    /**
     * Compute totals for a sale order
     * @param saleOrder The order to compute
     * @return The updated sale order with computed totals
     * @throws AxelorException if computation fails
     */
    SaleOrder computeTotals(SaleOrder saleOrder) throws AxelorException;

    /**
     * Validate a sale order before confirmation
     * @param saleOrder The order to validate
     * @throws AxelorException if validation fails
     */
    void validate(SaleOrder saleOrder) throws AxelorException;

    /**
     * Confirm a sale order and transition to confirmed status
     * @param saleOrder The order to confirm
     * @return The confirmed sale order
     * @throws AxelorException if confirmation fails
     */
    SaleOrder confirm(SaleOrder saleOrder) throws AxelorException;

    /**
     * Check if order can be confirmed
     * @param saleOrder The order to check
     * @return true if order can be confirmed
     */
    boolean canConfirm(SaleOrder saleOrder);
}
```

## Service Implementation

### Basic Service Implementation with @Transactional

```java
package com.axelor.apps.sale.service;

import com.axelor.apps.base.db.Company;
import com.axelor.apps.base.db.Partner;
import com.axelor.apps.base.service.app.AppBaseService;
import com.axelor.apps.sale.db.SaleOrder;
import com.axelor.apps.sale.db.SaleOrderLine;
import com.axelor.apps.sale.db.repo.SaleOrderRepository;
import com.axelor.apps.sale.exception.SaleExceptionMessage;
import com.axelor.exception.AxelorException;
import com.axelor.exception.db.repo.TraceBackRepository;
import com.axelor.i18n.I18n;
import com.axelor.inject.Beans;
import com.google.inject.Inject;
import com.google.inject.persist.Transactional;
import java.lang.invoke.MethodHandles;
import java.math.BigDecimal;
import java.math.RoundingMode;
import java.time.LocalDate;
import java.util.ArrayList;
import java.util.List;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;


public class SaleOrderServiceImpl implements SaleOrderService {

    private static final Logger LOG = LoggerFactory.getLogger(MethodHandles.lookup().lookupClass());

    protected SaleOrderRepository saleOrderRepository;
    protected SaleOrderLineService saleOrderLineService;
    protected SaleOrderComputeService saleOrderComputeService;
    protected AppBaseService appBaseService;

    @Inject
    public SaleOrderServiceImpl(
            SaleOrderRepository saleOrderRepository,
            SaleOrderLineService saleOrderLineService,
            SaleOrderComputeService saleOrderComputeService,
            AppBaseService appBaseService) {
        this.saleOrderRepository = saleOrderRepository;
        this.saleOrderLineService = saleOrderLineService;
        this.saleOrderComputeService = saleOrderComputeService;
        this.appBaseService = appBaseService;
    }

    @Override
    @Transactional(rollbackOn = {Exception.class})
    public SaleOrder createOrder(Partner customer) throws AxelorException {
        LOG.debug("Creating sale order for customer: {}", customer.getName());

        if (customer == null) {
            throw new AxelorException(
                TraceBackRepository.CATEGORY_MISSING_FIELD,
                I18n.get(SaleExceptionMessage.SALE_ORDER_CUSTOMER_REQUIRED)
            );
        }

        SaleOrder saleOrder = new SaleOrder();
        saleOrder.setCustomer(customer);
        saleOrder.setCompany(customer.getMainCompany());
        saleOrder.setCreationDate(appBaseService.getTodayDate(saleOrder.getCompany()));
        saleOrder.setOrderDate(saleOrder.getCreationDate());
        saleOrder.setStatusSelect(SaleOrderRepository.STATUS_DRAFT_QUOTATION);
        saleOrder.setCurrency(customer.getCurrency());
        saleOrder.setPriceList(customer.getSalePriceList());
        saleOrder.setPaymentMode(customer.getPaymentMode());
        saleOrder.setPaymentCondition(customer.getPaymentCondition());
        saleOrder.setDeliveryAddress(customer.getMainAddress());
        saleOrder.setSaleOrderLineList(new ArrayList<>());

        saleOrder = saleOrderRepository.save(saleOrder);
        LOG.debug("Sale order created with ID: {}", saleOrder.getId());

        return saleOrder;
    }

    @Override
    @Transactional(rollbackOn = {Exception.class})
    public SaleOrder computeTotals(SaleOrder saleOrder) throws AxelorException {
        LOG.debug("Computing totals for sale order: {}", saleOrder.getId());

        if (saleOrder == null) {
            throw new AxelorException(
                TraceBackRepository.CATEGORY_MISSING_FIELD,
                I18n.get(SaleExceptionMessage.SALE_ORDER_REQUIRED)
            );
        }

        saleOrder = saleOrderComputeService.computeLines(saleOrder);
        saleOrder = saleOrderComputeService.computeTaxes(saleOrder);
        saleOrder = saleOrderComputeService.computeTotals(saleOrder);

        saleOrder = saleOrderRepository.save(saleOrder);
        LOG.debug("Totals computed. ExTaxTotal: {}, InTaxTotal: {}",
            saleOrder.getExTaxTotal(), saleOrder.getInTaxTotal());

        return saleOrder;
    }

    @Override
    public void validate(SaleOrder saleOrder) throws AxelorException {
        LOG.debug("Validating sale order: {}", saleOrder.getId());

        if (saleOrder.getCustomer() == null) {
            throw new AxelorException(
                TraceBackRepository.CATEGORY_MISSING_FIELD,
                I18n.get(SaleExceptionMessage.SALE_ORDER_CUSTOMER_REQUIRED)
            );
        }

        if (saleOrder.getCompany() == null) {
            throw new AxelorException(
                TraceBackRepository.CATEGORY_MISSING_FIELD,
                I18n.get(SaleExceptionMessage.SALE_ORDER_COMPANY_REQUIRED)
            );
        }

        if (saleOrder.getSaleOrderLineList() == null ||
            saleOrder.getSaleOrderLineList().isEmpty()) {
            throw new AxelorException(
                TraceBackRepository.CATEGORY_INCONSISTENCY,
                I18n.get(SaleExceptionMessage.SALE_ORDER_NO_LINES)
            );
        }

        for (SaleOrderLine line : saleOrder.getSaleOrderLineList()) {
            if (line.getProduct() == null) {
                throw new AxelorException(
                    TraceBackRepository.CATEGORY_MISSING_FIELD,
                    I18n.get(SaleExceptionMessage.SALE_ORDER_LINE_PRODUCT_REQUIRED)
                );
            }

            if (line.getQty() == null || line.getQty().compareTo(BigDecimal.ZERO) <= 0) {
                throw new AxelorException(
                    TraceBackRepository.CATEGORY_INCONSISTENCY,
                    I18n.get(SaleExceptionMessage.SALE_ORDER_LINE_INVALID_QTY),
                    line.getProduct().getName()
                );
            }
        }

        LOG.debug("Sale order validation successful");
    }

    @Override
    @Transactional(rollbackOn = {Exception.class})
    public SaleOrder confirm(SaleOrder saleOrder) throws AxelorException {
        LOG.info("Confirming sale order: {}", saleOrder.getId());

        validate(saleOrder);

        if (!canConfirm(saleOrder)) {
            throw new AxelorException(
                TraceBackRepository.CATEGORY_INCONSISTENCY,
                I18n.get(SaleExceptionMessage.SALE_ORDER_CANNOT_CONFIRM)
            );
        }

        saleOrder.setStatusSelect(SaleOrderRepository.STATUS_ORDER_CONFIRMED);
        saleOrder.setConfirmationDateTime(
            appBaseService.getTodayDateTime(saleOrder.getCompany()).toLocalDateTime()
        );
        saleOrder.setConfirmedByUser(AuthUtils.getUser());

        if (saleOrder.getOrderNumber() == null) {
            saleOrder.setOrderNumber(
                Beans.get(SequenceService.class)
                    .getSequenceNumber("saleOrder", saleOrder.getCompany())
            );
        }

        saleOrder = saleOrderRepository.save(saleOrder);
        LOG.info("Sale order confirmed with number: {}", saleOrder.getOrderNumber());

        return saleOrder;
    }

    @Override
    public boolean canConfirm(SaleOrder saleOrder) {
        if (saleOrder == null) {
            return false;
        }

        Integer status = saleOrder.getStatusSelect();
        return status != null &&
               (status.equals(SaleOrderRepository.STATUS_DRAFT_QUOTATION) ||
                status.equals(SaleOrderRepository.STATUS_FINALIZED_QUOTATION));
    }
}
```

## Dependency Injection with Google Guice

### CRITICAL: DO NOT Use @Singleton on Services

**NEVER use `@Singleton` annotation on service implementations** unless you have explicit business justification documented in specifications.

```java
// WRONG - Do NOT use @Singleton without documented justification
import com.google.inject.Singleton;

@Singleton
public class SaleOrderServiceImpl implements SaleOrderService {
  // ...
}

// CORRECT - No @Singleton (prototype scope is default)
public class SaleOrderServiceImpl implements SaleOrderService {
  // ...
}
```

### Constructor Injection (MANDATORY)

```java
package com.axelor.apps.sale.service;

import com.google.inject.Inject;

public class SaleOrderServiceImpl implements SaleOrderService {

    protected final SaleOrderRepository saleOrderRepository;
    protected final SaleOrderLineService saleOrderLineService;
    protected final AppBaseService appBaseService;

    @Inject
    public SaleOrderServiceImpl(
            SaleOrderRepository saleOrderRepository,
            SaleOrderLineService saleOrderLineService,
            AppBaseService appBaseService) {
        this.saleOrderRepository = saleOrderRepository;
        this.saleOrderLineService = saleOrderLineService;
        this.appBaseService = appBaseService;
    }
}
```

### Field Injection - DO NOT USE

> **WARNING:** Field injection (`@Inject` on fields) is **NOT recommended** in Axelor services.
> Always use **constructor injection** instead.
```java
// DO NOT DO THIS
public class SaleOrderServiceImpl implements SaleOrderService {

    @Inject
    protected SaleOrderRepository saleOrderRepository;  // BAD

    @Inject
    protected SaleOrderLineService saleOrderLineService;  // BAD
}

// DO THIS INSTEAD - Constructor injection
public class SaleOrderServiceImpl implements SaleOrderService {

    private final SaleOrderRepository saleOrderRepository;
    private final SaleOrderLineService saleOrderLineService;

    @Inject
    public SaleOrderServiceImpl(
            SaleOrderRepository saleOrderRepository,
            SaleOrderLineService saleOrderLineService) {
        this.saleOrderRepository = saleOrderRepository;
        this.saleOrderLineService = saleOrderLineService;
    }
}
```

### Using Beans.get() for Runtime Resolution

```java
package com.axelor.apps.sale.service;
```

## Transaction Management

### @Transactional Annotation

```java
package com.axelor.apps.sale.service;

import com.google.inject.persist.Transactional;

public class SaleOrderServiceImpl implements SaleOrderService {

    // Basic transaction - commits on success, rolls back on exception
    @Transactional
    public SaleOrder create(SaleOrder saleOrder) {
        return saleOrderRepository.save(saleOrder);
    }

    // Explicit rollback on all exceptions
    @Transactional(rollbackOn = {Exception.class})
    public SaleOrder confirm(SaleOrder saleOrder) throws AxelorException {
        validate(saleOrder);
        saleOrder.setStatusSelect(SaleOrderRepository.STATUS_ORDER_CONFIRMED);
        return saleOrderRepository.save(saleOrder);
    }

    // Read-only transaction (optimization for queries)
    @Transactional(readOnly = true)
    public List<SaleOrder> findPendingOrders() {
        return saleOrderRepository.all()
            .filter("self.statusSelect = ?", SaleOrderRepository.STATUS_DRAFT_QUOTATION)
            .fetch();
    }

    // No transaction annotation - for pure business logic without persistence
    public BigDecimal calculateDiscount(SaleOrder saleOrder) {
        BigDecimal total = saleOrder.getExTaxTotal();
        BigDecimal discountPercent = saleOrder.getDiscountPercent();
        return total.multiply(discountPercent).divide(BigDecimal.valueOf(100), 2, RoundingMode.HALF_UP);
    }
}
```

### Transaction Propagation Patterns

```java
package com.axelor.apps.sale.service;

import com.google.inject.persist.Transactional;
import com.google.inject.Inject;

public class SaleOrderWorkflowServiceImpl implements SaleOrderWorkflowService {

    private final SaleOrderService saleOrderService;
    private final InvoiceService invoiceService;
    private final StockMoveService stockMoveService;

    @Inject
    public SaleOrderWorkflowServiceImpl(
            SaleOrderService saleOrderService,
            InvoiceService invoiceService,
            StockMoveService stockMoveService) {
        this.saleOrderService = saleOrderService;
        this.invoiceService = invoiceService;
        this.stockMoveService = stockMoveService;
    }

    // Main transaction coordinates multiple operations
    @Transactional(rollbackOn = {Exception.class})
    public SaleOrder confirmAndGenerate(SaleOrder saleOrder) throws AxelorException {
        // All these operations run in the same transaction
        saleOrder = saleOrderService.confirm(saleOrder);

        if (saleOrder.getGenerateInvoice()) {
            invoiceService.generateFromOrder(saleOrder);
        }

        if (saleOrder.getGenerateDelivery()) {
            stockMoveService.generateFromOrder(saleOrder);
        }

        return saleOrder;
    }

    // Nested transaction - if this fails, parent can continue
    @Transactional(rollbackOn = {Exception.class})
    public void sendNotification(SaleOrder saleOrder) {
        try {
            EmailService emailService = Beans.get(EmailService.class);
            emailService.sendOrderConfirmation(saleOrder);
        } catch (Exception e) {
            LOG.error("Failed to send notification", e);
            // Don't propagate - notification failure shouldn't fail the order
        }
    }
}
```

### Validation Methods

```java
public class SaleOrderServiceImpl implements SaleOrderService {

    @Override
    public void validate(SaleOrder saleOrder) throws AxelorException {
        validateRequiredFields(saleOrder);
        validateBusinessRules(saleOrder);
        validateLines(saleOrder);
    }

    protected void validateRequiredFields(SaleOrder saleOrder) throws AxelorException {
        if (saleOrder.getCustomer() == null) {
            throw new AxelorException(
                TraceBackRepository.CATEGORY_MISSING_FIELD,
                I18n.get(SaleExceptionMessage.SALE_ORDER_CUSTOMER_REQUIRED)
            );
        }

        if (saleOrder.getCompany() == null) {
            throw new AxelorException(
                TraceBackRepository.CATEGORY_MISSING_FIELD,
                I18n.get(SaleExceptionMessage.SALE_ORDER_COMPANY_REQUIRED)
            );
        }
    }

    protected void validateBusinessRules(SaleOrder saleOrder) throws AxelorException {
        if (saleOrder.getExTaxTotal().compareTo(BigDecimal.ZERO) <= 0) {
            throw new AxelorException(
                TraceBackRepository.CATEGORY_INCONSISTENCY,
                I18n.get(SaleExceptionMessage.SALE_ORDER_INVALID_TOTAL)
            );
        }

        if (saleOrder.getCustomer().getBlocked()) {
            throw new AxelorException(
                TraceBackRepository.CATEGORY_INCONSISTENCY,
                I18n.get(SaleExceptionMessage.SALE_ORDER_CUSTOMER_BLOCKED)
            );
        }
    }

    protected void validateLines(SaleOrder saleOrder) throws AxelorException {
        if (saleOrder.getSaleOrderLineList() == null ||
            saleOrder.getSaleOrderLineList().isEmpty()) {
            throw new AxelorException(
                TraceBackRepository.CATEGORY_INCONSISTENCY,
                I18n.get(SaleExceptionMessage.SALE_ORDER_NO_LINES)
            );
        }

        for (SaleOrderLine line : saleOrder.getSaleOrderLineList()) {
            saleOrderLineService.validate(line);
        }
    }
}
```

### Computation Methods

```java
public class SaleOrderComputeServiceImpl implements SaleOrderComputeService {

    @Transactional
    public SaleOrder computeLines(SaleOrder saleOrder) throws AxelorException {
        for (SaleOrderLine line : saleOrder.getSaleOrderLineList()) {
            line = saleOrderLineService.compute(line, saleOrder);
        }
        return saleOrder;
    }

    @Transactional
    public SaleOrder computeTaxes(SaleOrder saleOrder) throws AxelorException {
        Map<TaxLine, BigDecimal> taxMap = new HashMap<>();

        for (SaleOrderLine line : saleOrder.getSaleOrderLineList()) {
            TaxLine taxLine = line.getTaxLine();
            if (taxLine != null) {
                BigDecimal taxAmount = line.getExTaxTotal()
                    .multiply(taxLine.getValue())
                    .divide(BigDecimal.valueOf(100), 2, RoundingMode.HALF_UP);

                taxMap.merge(taxLine, taxAmount, BigDecimal::add);
            }
        }

        // Clear existing tax lines
        saleOrder.getSaleOrderTaxLineList().clear();

        // Add new tax lines
        for (Map.Entry<TaxLine, BigDecimal> entry : taxMap.entrySet()) {
            SaleOrderTaxLine taxLine = new SaleOrderTaxLine();
            taxLine.setSaleOrder(saleOrder);
            taxLine.setTaxLine(entry.getKey());
            taxLine.setTaxAmount(entry.getValue());
            saleOrder.addSaleOrderTaxLineListItem(taxLine);
        }

        return saleOrder;
    }

    @Transactional
    public SaleOrder computeTotals(SaleOrder saleOrder) {
        BigDecimal exTaxTotal = BigDecimal.ZERO;
        BigDecimal taxTotal = BigDecimal.ZERO;

        for (SaleOrderLine line : saleOrder.getSaleOrderLineList()) {
            exTaxTotal = exTaxTotal.add(line.getExTaxTotal());
        }

        for (SaleOrderTaxLine taxLine : saleOrder.getSaleOrderTaxLineList()) {
            taxTotal = taxTotal.add(taxLine.getTaxAmount());
        }

        saleOrder.setExTaxTotal(exTaxTotal);
        saleOrder.setTaxTotal(taxTotal);
        saleOrder.setInTaxTotal(exTaxTotal.add(taxTotal));

        return saleOrder;
    }
}
```

### Check Methods

```java
public class SaleOrderServiceImpl implements SaleOrderService {

    @Override
    public boolean canConfirm(SaleOrder saleOrder) {
        return saleOrder != null
            && saleOrder.getStatusSelect() != null
            && (saleOrder.getStatusSelect().equals(SaleOrderRepository.STATUS_DRAFT_QUOTATION)
                || saleOrder.getStatusSelect().equals(SaleOrderRepository.STATUS_FINALIZED_QUOTATION))
            && saleOrder.getExTaxTotal().compareTo(BigDecimal.ZERO) > 0;
    }

    @Override
    public boolean canCancel(SaleOrder saleOrder) {
        return saleOrder != null
            && saleOrder.getStatusSelect() != null
            && !saleOrder.getStatusSelect().equals(SaleOrderRepository.STATUS_CANCELED)
            && !hasGeneratedInvoices(saleOrder)
            && !hasCompletedDeliveries(saleOrder);
    }

    @Override
    public boolean hasGeneratedInvoices(SaleOrder saleOrder) {
        return saleOrder.getInvoiceList() != null
            && !saleOrder.getInvoiceList().isEmpty();
    }
}
```

### Delete Methods

```java
public class SaleOrderServiceImpl implements SaleOrderService {

    @Transactional(rollbackOn = {Exception.class})
    public void delete(SaleOrder saleOrder) throws AxelorException {
        validateCanDelete(saleOrder);

        // Delete related entities first
        deleteRelatedLines(saleOrder);
        deleteRelatedTaxLines(saleOrder);

        // Delete the main entity
        saleOrderRepository.remove(saleOrder);

        LOG.info("Sale order deleted: {}", saleOrder.getId());
    }

    protected void validateCanDelete(SaleOrder saleOrder) throws AxelorException {
        if (saleOrder.getStatusSelect() >= SaleOrderRepository.STATUS_ORDER_CONFIRMED) {
            throw new AxelorException(
                TraceBackRepository.CATEGORY_INCONSISTENCY,
                I18n.get(SaleExceptionMessage.SALE_ORDER_CANNOT_DELETE_CONFIRMED)
            );
        }

        if (hasGeneratedInvoices(saleOrder)) {
            throw new AxelorException(
                TraceBackRepository.CATEGORY_INCONSISTENCY,
                I18n.get(SaleExceptionMessage.SALE_ORDER_CANNOT_DELETE_WITH_INVOICES)
            );
        }
    }

    protected void deleteRelatedLines(SaleOrder saleOrder) {
        if (saleOrder.getSaleOrderLineList() != null) {
            saleOrder.getSaleOrderLineList().clear();
        }
    }
}
```

## Exception Handling with AxelorException

### Basic Exception Handling

```java
public class SaleOrderServiceImpl implements SaleOrderService {

    @Override
    public SaleOrder confirm(SaleOrder saleOrder) throws AxelorException {
        try {
            validate(saleOrder);

            if (!canConfirm(saleOrder)) {
                throw new AxelorException(
                    TraceBackRepository.CATEGORY_INCONSISTENCY,
                    I18n.get(SaleExceptionMessage.SALE_ORDER_CANNOT_CONFIRM)
                );
            }

            return doConfirm(saleOrder);

        } catch (AxelorException e) {
            throw e;
        } catch (Exception e) {
            throw new AxelorException(
                e,
                TraceBackRepository.CATEGORY_CONFIGURATION_ERROR,
                I18n.get(SaleExceptionMessage.SALE_ORDER_CONFIRM_ERROR),
                saleOrder.getOrderNumber()
            );
        }
    }
}
```

### Exception Categories

```java
public class SaleOrderServiceImpl implements SaleOrderService {

    // Missing required field
    protected void checkCustomer(SaleOrder saleOrder) throws AxelorException {
        if (saleOrder.getCustomer() == null) {
            throw new AxelorException(
                TraceBackRepository.CATEGORY_MISSING_FIELD,
                I18n.get(SaleExceptionMessage.SALE_ORDER_CUSTOMER_REQUIRED)
            );
        }
    }

    // Business rule violation
    protected void checkBusinessRules(SaleOrder saleOrder) throws AxelorException {
        if (saleOrder.getCustomer().getBlocked()) {
            throw new AxelorException(
                TraceBackRepository.CATEGORY_INCONSISTENCY,
                I18n.get(SaleExceptionMessage.SALE_ORDER_CUSTOMER_BLOCKED),
                saleOrder.getCustomer().getName()
            );
        }
    }
```

## Batch Processing in Services

### Overview

When processing large datasets, services must use batch processing to avoid memory leaks.

### Batch Processing Service

```java
package com.axelor.apps.license.service;

import com.axelor.apps.base.helpers.Loggers;
import com.axelor.apps.license.db.License;
import com.axelor.apps.license.db.repo.LicenseRepository;
import com.axelor.db.JPA;
import com.axelor.db.Query;
import com.google.inject.Inject;
import com.google.inject.persist.Transactional;
import java.lang.invoke.MethodHandles;
import java.util.List;

public class LicenseBatchServiceImpl implements LicenseBatchService {

    private static final Logger LOG = LoggerFactory.getLogger(MethodHandles.lookup().lookupClass());
    private static final int BATCH_SIZE = 50;

    @Inject
    protected LicenseRepository licenseRepository;

    @Override
    public long processExpiredLicenses() {
        long processed = 0;

        Query<License> query = Query.of(License.class)
            .filter("self.endDate < :today AND self.status = :status")
            .bind("today", LocalDate.now())
            .bind("status", LicenseStatus.ACTIVE)
            .order("id");

        query.setMaxResults(BATCH_SIZE);
        List<License> licenses = query.fetch();

        while (!licenses.isEmpty()) {
            // Process batch
            for (License license : licenses) {
                try {
                    expireLicense(license);
                    processed++;
                } catch (Exception e) {
                    LOG.error("Error expiring license: {}", license.getId(), e);
                }
            }

            // CRITICAL: Clear JPA context to free memory
            JPA.clear();

            // Fetch next batch
            query.setFirstResult(query.getFirstResult() + BATCH_SIZE);
            licenses = query.fetch();
        }

        LOG.info("Processed {} expired licenses", processed);
        return processed;
    }

    @Transactional(rollbackOn = Exception.class)
    protected void expireLicense(License license) {
        license.setStatus(LicenseStatus.EXPIRED);
        license.setExpiredDate(LocalDate.now());
        licenseRepository.save(license);
    }
}
```

### Batch Processing with RequestScoper

For background jobs running outside HTTP request context:

```java
package com.axelor.apps.license.service;

import com.axelor.db.JPA;
import com.google.inject.servlet.RequestScoper;
import com.google.inject.servlet.ServletScopes;
import java.lang.invoke.MethodHandles;
import java.util.Collections;
import java.util.List;
import java.util.concurrent.Callable;

public class MassSaveServiceImpl implements Callable<Long> {

    private static final Logger LOG = LoggerFactory.getLogger(MethodHandles.lookup().lookupClass());
    private static final int BATCH_SIZE = 50;
    private List<Long> licenseIds;

    public MassSaveServiceImpl(List<Long> licenseIds) {
        this.licenseIds = licenseIds;
    }

    @Override
    public Long call() {
        // Create request scope for batch operation
        final RequestScoper scope = ServletScopes.scopeRequest(Collections.emptyMap());

        try (RequestScoper.CloseableScope ignored = scope.open()) {
            return process();
        }
    }

    private Long process() {
        long processed = 0;

        for (int i = 0; i < licenseIds.size(); i++) {
            Long licenseId = licenseIds.get(i);

            try {
                processLicense(licenseId);
                processed++;
            } catch (Exception e) {
                LOG.error("Error processing license: {}", licenseId, e);
            }

            // Clear JPA context every BATCH_SIZE
            if (i > 0 && i % BATCH_SIZE == 0) {
                JPA.clear();
            }
        }

        return processed;
    }

    @Transactional(rollbackOn = Exception.class)
    protected void processLicense(Long licenseId) {
        License license = licenseRepository.find(licenseId);
        if (license != null) {
            license.setProcessed(true);
            licenseRepository.save(license);
        }
    }
}
```

**Key Points:**
- Always call `JPA.clear()` between batches
- Use RequestScoper for background jobs
- Catch exceptions per-item to continue processing
- Log errors but don't stop batch
- Fixed batch size (typically 50-100)

**Pattern Frequency:** COMMON - Required for all bulk operations
