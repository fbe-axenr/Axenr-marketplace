# Repository Patterns in Axelor

## Overview

Repositories in Axelor are the data access layer built on top of JPA. They provide methods to query, persist, and manage domain entities.

### How Axelor Repositories Work

**Important:** Axelor automatically generates `[Entity]Repository` classes when you run `generateCode`. These generated repositories:
- Are named `[Entity]Repository` (e.g., `SaleOrderRepository`)
- Already extend the base `JpaRepository<Entity>` class
- Provide basic operations: `save()`, `remove()`, `find()`, `all()`
- Cannot be modified directly (they're regenerated)

**When you need customization:**
- Create a NEW class that extends the generated `[Entity]Repository`
- Common naming: `[Entity]Repo` extends `[Entity]Repository` (real Axelor convention)
- Use constructor injection with @Inject for services/repositories
- Add your custom query methods and business logic there

### Repository Hierarchy

```
JpaRepository<T>                    (Axelor base class)
    ↓
[Entity]Repository                  (Auto-generated, DO NOT MODIFY, in build/src-gen/)
    ↓
[Entity]Repo                        (Custom, only if needed, in src/main/java/)
```

## Custom Repository (When Needed)

**Only create this when you need custom behavior:**

```java
package com.axelor.apps.sale.db.repo;

import com.axelor.apps.sale.db.SaleOrder;
import com.axelor.apps.sale.service.SaleOrderService;
import com.axelor.db.Query;
import com.axelor.inject.Beans;
import com.google.inject.Inject;
import java.io.IOException;
import java.time.LocalDate;
import java.util.List;

// Extends the AUTO-GENERATED SaleOrderRepository (common naming: [Entity]Repo)
public class SaleOrderRepo extends SaleOrderRepository {

    protected SaleOrderService saleOrderService;

    @Inject
    public SaleOrderRepo(SaleOrderService saleOrderService) {
        this.saleOrderService = saleOrderService;
    }

    // Override save() with business logic and service calls
    @Override
    public SaleOrder save(SaleOrder saleOrder) {
        // Call service for business logic before save
        if (saleOrder.getCustomer() != null) {
            saleOrder.getCustomer().setHasActiveOrders(true);
        }

        // First save to persist relationships
        saleOrder = super.save(saleOrder);

        try {
            // Call service to compute derived fields
            saleOrder.setFullName(saleOrderService.computeFullName(saleOrder));
            saleOrder.setTotalTax(saleOrderService.computeTotalTax(saleOrder));
        } catch (Exception e) {
            ExceptionHelper.trace(e);
        }

        // Second save to persist computed fields
        return super.save(saleOrder);
    }

    // Override remove() for cleanup logic
    @Override
    public void remove(SaleOrder saleOrder) {
        // Clean up related data before removal
        if (saleOrder.getCustomer() != null) {
            Long remainingOrders = all()
                .filter("self.customer = :customer AND self.id != :orderId")
                .bind("customer", saleOrder.getCustomer())
                .bind("orderId", saleOrder.getId())
                .count();

            if (remainingOrders == 0) {
                saleOrder.getCustomer().setHasActiveOrders(false);
            }
        }

        super.remove(saleOrder);
    }

    // Add custom query methods
    public List<SaleOrder> findByCustomer(Long customerId) {
        return all()
            .filter("self.customer.id = :customerId")
            .bind("customerId", customerId)
            .fetch();
    }

    public long countByStatus(int status) {
        return all()
            .filter("self.statusSelect = :status")
            .bind("status", status)
            .count();
    }
}
```

## Native SQL Queries

### When to Use Native SQL

Native SQL should be used sparingly and only when:
- Complex aggregations not possible with JPQL
- Database-specific functions required
- Performance optimization for complex queries
- Bulk operations

```java
public class SaleOrderRepo extends SaleOrderRepository {

    // Native SQL for complex aggregation
    public List<Object[]> getOrderStatsByMonth(Company company, int year) {
        String sql = "SELECT " +
                     "EXTRACT(MONTH FROM order_date) as month, " +
                     "COUNT(*) as order_count, " +
                     "SUM(ex_tax_total) as total_amount, " +
                     "AVG(ex_tax_total) as avg_amount " +
                     "FROM sale_order " +
                     "WHERE company_id = ?1 " +
                     "AND EXTRACT(YEAR FROM order_date) = ?2 " +
                     "GROUP BY EXTRACT(MONTH FROM order_date) " +
                     "ORDER BY month";

        javax.persistence.Query query = getEntityManager()
            .createNativeQuery(sql)
            .setParameter(1, company.getId())
            .setParameter(2, year);

        return query.getResultList();
    }

    // Native SQL for bulk update
    public int bulkUpdateStatusForOldDrafts(LocalDate beforeDate) {
        String sql = "UPDATE sale_order " +
                     "SET status_select = ?1 " +
                     "WHERE status_select = ?2 " +
                     "AND creation_date < ?3";

        return getEntityManager()
            .createNativeQuery(sql)
            .setParameter(1, STATUS_CANCELED)
            .setParameter(2, STATUS_DRAFT_QUOTATION)
            .setParameter(3, beforeDate)
            .executeUpdate();
    }

    // Native SQL with entity result
    public List<SaleOrder> findUsingNativeQuery(Company company) {
        String sql = "SELECT * FROM sale_order " +
                     "WHERE company_id = ?1 " +
                     "ORDER BY order_date DESC";

        return getEntityManager()
            .createNativeQuery(sql, SaleOrder.class)
            .setParameter(1, company.getId())
            .getResultList();
    }
}
```

## Repository Best Practices

### 1. Use Constants for Magic Values

```java
public class SaleOrderRepo extends SaleOrderRepository {

    // Good: Use constants
    public static final int STATUS_DRAFT = 1;
    public static final int STATUS_CONFIRMED = 2;

    public List<SaleOrder> findDraftOrders() {
        return all()
            .filter("self.statusSelect = ?1", STATUS_DRAFT)
            .fetch();
    }

    // Bad: Magic numbers
    public List<SaleOrder> findConfirmedOrders() {
        return all()
            .filter("self.statusSelect = ?1", 2)  // What is 2?
            .fetch();
    }
}
```

### 2. Use Named Parameters for Clarity

```java
public class SaleOrderRepo extends SaleOrderRepository {

    // Good: Named parameters
    public List<SaleOrder> findByCustomerAndStatus(Partner customer, Integer status) {
        return all()
            .filter("self.customer = :customer AND self.statusSelect = :status")
            .bind("customer", customer)

    // Less clear: Positional parameters

            .fetch();
```

### 3. Avoid N+1 Query Problems

```java
public class SaleOrderRepo extends SaleOrderRepository {

    // Good: Use fetch join
    public List<SaleOrder> findAllWithLinesEager() {
        String jpql = "SELECT DISTINCT self FROM SaleOrder self " +
                      "LEFT JOIN FETCH self.saleOrderLineList";

        return all()
            .filter(jpql)
            .fetch();
    }

    // Bad: Will cause N+1 queries when accessing lines
    public List<SaleOrder> findAllLazy() {
        return all().fetch();
        // Accessing saleOrderLineList later will trigger additional queries
    }
}
```

## Overriding save() for Computed Fields

### Overview

Repository classes can override the `save()` method to automatically compute derived fields before persistence. This is common for generating names, codes, or maintaining denormalized data.

### Basic save() Override Pattern

```java
package com.axelor.apps.license.db.repo;

import com.axelor.apps.license.db.License;
import com.axelor.db.JPA;
import java.util.StringJoiner;

// Custom repository extending the AUTO-GENERATED LicenseRepository
public class LicenseRepo extends LicenseRepository {

    @Override
    public License save(License entity) {
        // First save to get ID and relationships
        License license = super.save(entity);

        // Compute derived name field
        StringJoiner stringJoiner = new StringJoiner(" - ");
        stringJoiner.add(license.getApplication().getName())
                    .add(license.getLicenseUsage().getName());

        if (license.getLicenseScope() != null) {
            stringJoiner.add(license.getLicenseScope().getName());
        }

        // Set computed field
        license.setName(stringJoiner.toString());

        // Second save with computed values
        return JPA.save(license);
    }
}
```

**Key Points:**
- Call `super.save()` first to persist entity with ID
- Compute derived fields using related entities
- Call `JPA.save()` to persist computed values
- Double-save pattern ensures ID exists for relationships

### save() Override with Multiple Computed Fields

```java
package com.axelor.apps.license.db.repo;

import com.axelor.apps.license.db.LicenseApplication;
import com.axelor.db.JPA;
import java.util.StringJoiner;

// Custom repository extending the AUTO-GENERATED LicenseApplicationRepository
public class LicenseApplicationRepo extends LicenseApplicationRepository {

    @Override
    public LicenseApplication save(LicenseApplication entity) {
        // Initial save
        LicenseApplication application = super.save(entity);

        // Compute application code
        if (application.getApplicationCode() == null) {
            application.setApplicationCode(generateCode(application));
        }

        // Compute full name
        StringJoiner nameBuilder = new StringJoiner(" - ");
        if (application.getIntegrator() != null) {
            nameBuilder.add(application.getIntegrator().getName());
        }
        if (application.getCustomer() != null) {
            nameBuilder.add(application.getCustomer().getName());
        }
        nameBuilder.add(application.getApplicationName());

        application.setFullName(nameBuilder.toString());

        // Compute license count
        if (application.getLicenses() != null) {
            application.setActiveLicenseCount(
                (int) application.getLicenses().stream()
                    .filter(lic -> lic.getStatus() == LicenseStatus.ACTIVE)
                    .count()
            );
        }

        // Final save with all computed values
        return JPA.save(application);
    }

    private String generateCode(LicenseApplication application) {
        String prefix = application.getIntegrator() != null
            ? application.getIntegrator().getCode()
            : "GEN";

        long count = Query.of(LicenseApplication.class)
            .filter("self.integrator = :integrator")
            .bind("integrator", application.getIntegrator())
            .count();

        return String.format("%s-%04d", prefix, count + 1);
    }
}
```

### save() Override with Validation

```java
package com.axelor.apps.license.db.repo;

import com.axelor.apps.license.db.License;
import com.axelor.db.JPA;
import com.axelor.exception.AxelorException;
import com.axelor.exception.db.repo.TraceBackRepository;
import com.axelor.i18n.I18n;

// Custom repository extending the AUTO-GENERATED LicenseRepository
public class LicenseRepo extends LicenseRepository {

    @Override
    public License save(License entity) {
        // Validate before save
        validateLicense(entity);

        // First save
        License license = super.save(entity);

        // Compute name
        license.setName(computeName(license));

        // Update denormalized fields
        updateDenormalizedFields(license);

        // Final save
        return JPA.save(license);
    }

    private void validateLicense(License license) {
        if (license.getEndDate() != null && license.getIssueDate() != null) {
            if (license.getEndDate().isBefore(license.getIssueDate())) {
                throw new IllegalArgumentException(
                    I18n.get("End date must be after issue date")
                );
            }
        }
    }

    private String computeName(License license) {
        StringJoiner joiner = new StringJoiner(" - ");
        joiner.add(license.getApplication().getName());
        joiner.add(license.getLicenseUsage().getName());

        if (license.getLicenseScope() != null) {
            joiner.add(license.getLicenseScope().getName());
        }

        return joiner.toString();
    }

    private void updateDenormalizedFields(License license) {
        // Copy customer data for performance
        if (license.getCustomer() != null) {
            license.setCustomerName(license.getCustomer().getName());
            license.setCustomerEmail(license.getCustomer().getEmail());
        }

        // Copy application data
        if (license.getApplication() != null) {
            license.setApplicationName(license.getApplication().getApplicationName());
        }
    }
}
```
